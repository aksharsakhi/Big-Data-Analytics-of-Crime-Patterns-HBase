\begin{frame}{Application Introduction, Problem Statement \& Paradigm Shift}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \textbf{\textcolor{NavyBlue}{Review 1 $\to$ Review 2 Paradigm Shift:}}
            \vspace{0.1cm}
            \begin{itemize}\scriptsize
                \item \textbf{Review 1 (Batch OLAP):} MapReduce \& Hive -- high-latency batch aggregations; no random read/write access.
                \item \textbf{Review 2 (Real-Time NoSQL):} \textbf{Apache HBase} on HDFS -- sub-second point lookups (\texttt{GET}), selective column scans, and prefix filtering.
            \end{itemize}
            \vspace{0.15cm}
            \textbf{\textcolor{NavyBlue}{Domain:}} \textit{Urban Public Safety \& Law Enforcement Intelligence.}
        \end{column}
        \begin{column}{0.48\textwidth}
            \textbf{\textcolor{AmritaMaroon}{Core Challenges in Urban Policing:}}
            \vspace{0.1cm}
            \begin{itemize}\scriptsize
                \item City police departments generate tens of thousands of incident reports weekly.
                \item RDBMS systems suffer write contention and lock degradation under streaming loads.
                \item Standard HDFS batch files cannot provide sub-second point lookups for active investigations.
            \end{itemize}
            \vspace{0.15cm}
            \textbf{\textcolor{AmritaMaroon}{Why Apache HBase?}}
            \begin{itemize}\scriptsize
                \item Wide-column distributed NoSQL database on HDFS.
                \item Real-time random read/write access to millions of crime incidents.
                \item Anti-hotspotting composite row key distributes writes evenly across regions.
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

            \centering
            \includegraphics[width=0.92\textwidth]{images/rowkey_design.png}
            \vspace{0.05cm}
            {\tiny\textit{Row Key: \texttt{District\#CrimeType\#IncidentID}}}
            \vspace{0.08cm}
            \begin{itemize}\fontsize{6}{7.5}\selectfont
                \item \textbf{Anti-Hotspotting:} District code salting distributes writes across RegionServers (avoids monotonic timestamp hotspot).
                \item \textbf{Prefix-Scannable:} Fast jurisdiction scans via \texttt{PrefixFilter('011\#')}.
                \item \textbf{Categorical Drill-Down:} Crime-type embedded in key enables \texttt{RowFilter(regex)} pattern matching.
                \item \textbf{Uniqueness:} Incident ID suffix guarantees collision-free row keys.
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}[fragile,shrink=5]{Step 5: HBase Java API Implementation \& Live Output}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            {\scriptsize\textbf{Class:} \texttt{bigdata.HBaseCrimeOperations.java} -- Table Admin, Put, Get, Filter Scan, Delete.}
            \begin{lstlisting}[language=Java]
Configuration config = HBaseConfiguration.create();
try (Connection conn = 
     ConnectionFactory.createConnection(config);
     Table table = conn.getTable(
       TableName.valueOf("crime_records"))) {
    
    // Filtered Scan: Arrest tracking
    Scan scan = new Scan();
    SingleColumnValueFilter filter = 
      new SingleColumnValueFilter(
        Bytes.toBytes("details"), 
        Bytes.toBytes("arrest"),
        CompareOperator.EQUAL, 
        new BinaryComparator(
          Bytes.toBytes("true")));
    filter.setFilterIfMissing(true);
    scan.setFilter(filter);
    scan.setLimit(5);

    try (ResultScanner scanner = 
         table.getScanner(scan)) {
        for (Result r : scanner) {
            System.out.println("Match: " 
              + Bytes.toString(r.getRow()));
        }
    }
}
            \end{lstlisting}
