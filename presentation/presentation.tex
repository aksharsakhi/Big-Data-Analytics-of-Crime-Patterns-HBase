\begin{frame}{Evolution: Review 1 (Batch) $\to$ Review 2 (Real-Time NoSQL)}
    \begin{columns}
        \begin{column}{0.48\textwidth}
            \begin{block}{\textbf{Review 1: Batch OLAP Processing}}
                \begin{itemize}
                    \item \textbf{Engines:} Hadoop MapReduce \& Apache Hive.
                    \item \textbf{Access Pattern:} High-latency batch aggregation over historical records.
                    \item \textbf{Limitation:} Lacks random read/write capability; slow table scans for point lookups.
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{block}{\textbf{Review 2: Real-Time Operational NoSQL}}
                \begin{itemize}
                    \item \textbf{Engine:} \textbf{Apache HBase} on HDFS.
                    \item \textbf{Access Pattern:} Sub-second point lookups (\texttt{GET}), selective column scans, and prefix filtering.
                    \item \textbf{Target:} 911 dispatch, beat officer surveillance, and real-time arrest tracking.
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}

% Slide 3: Step 1 - Problem Statement & Motivation
\begin{frame}{Step 1: Application Introduction \& Problem Statement}
    \begin{itemize}
        \item \textbf{Domain:} Urban Public Safety \& Law Enforcement Intelligence.
        \item \textbf{The Core Challenge:}
        \begin{itemize}
            \item City police departments generate tens of thousands of incident reports weekly.
            \item RDBMS systems suffer write contention and lock degradation under streaming loads.
            \item Standard HDFS batch files cannot provide sub-second point lookups for active investigations.
        \end{itemize}
        \item \textbf{Our Solution with Apache HBase:}
        \begin{itemize}
            \item Wide-column distributed NoSQL database on HDFS.
            \item Real-time, random read/write access to millions of crime incidents.
            \item Anti-hotspotting composite row key to distribute writes evenly across regions.
        \end{itemize}
    \end{itemize}
\end{frame}

% Slide 6: Step 3 - Composite Row Key Engineering
\begin{frame}{Step 3: Composite Row Key Engineering}
    \begin{center}
        \includegraphics[width=0.75\textwidth]{images/rowkey_design.png}
    \end{center}
    \vspace{-0.2cm}
    \begin{columns}
        \begin{column}{0.48\textwidth}
            \textbf{Why NOT auto-increment / timestamp alone?}
            \begin{itemize}
                \item Monotonically increasing keys cause \textbf{region hotspotting}.
                \item All write traffic converges on a single RegionServer.
            \end{itemize}
        \end{column}
        \begin{column}{0.48\textwidth}
            \textbf{Our Row Key Benefits:}
            \begin{itemize}
                \item \textbf{District Code Salting:} Balances load across nodes.
                \item \textbf{Prefix-Scannable:} Fast scans via \texttt{PrefixFilter('011\#')}.
                \item \textbf{Categorical Drill-Down:} Scans by crime type.
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

% Slide 9: Step 4 - HBase Shell Filters (Part 1)
\begin{frame}[fragile]{Step 4: HBase Analytical Filters (Rubric: 4 Marks)}
    \begin{block}{\textbf{Filter 1: SingleColumnValueFilter -- Arrest Tracking}}
        \textit{Goal: Identify crimes where suspects were successfully apprehended.}
        \begin{lstlisting}[language=bash]
scan 'crime_records', {FILTER => "SingleColumnValueFilter('details', 'arrest', =, 'binary:true')", LIMIT => 5}
        \end{lstlisting}
    \end{block}

    \begin{block}{\textbf{Filter 2: SingleColumnValueFilter -- Domestic Violence Tracking}}
        \textit{Goal: Filter domestic conflict incidents for family safety intervention units.}
        \begin{lstlisting}[language=bash]
scan 'crime_records', {FILTER => "SingleColumnValueFilter('details', 'domestic', =, 'binary:true')", LIMIT => 5}
        \end{lstlisting}
    \end{block}

    \begin{block}{\textbf{Filter 3: PrefixFilter -- Rapid Jurisdiction Retrieval}}
        \textit{Goal: Retrieve all crime incidents logged in District 019 jurisdiction.}
        \begin{lstlisting}[language=bash]
scan 'crime_records', {FILTER => "PrefixFilter('019#')", LIMIT => 5}
        \end{lstlisting}
    \end{block}
\end{frame}

% Slide 10: Step 4 - HBase Shell Filters (Part 2)
\begin{frame}[fragile]{Step 4: HBase Analytical Filters (Continued)}
    \begin{block}{\textbf{Filter 4: RowFilter with RegexStringComparator -- Narcotics Offenses}}
        \textit{Goal: Trace narcotics cases across all districts using row key pattern matching.}
        \begin{lstlisting}[language=bash]
scan 'crime_records', {FILTER => "RowFilter(=, 'regexstring:^.*#NARCOTICS#.*')", LIMIT => 5}
        \end{lstlisting}
    \end{block}

    \begin{block}{\textbf{Filter 5: ValueFilter with SubstringComparator -- Vehicle Incidents}}
        \textit{Goal: Full-value text search for incidents mentioning vehicle offenses.}
        \begin{lstlisting}[language=bash]
scan 'crime_records', {FILTER => "ValueFilter(=, 'substring:VEHICLE')", LIMIT => 5}
        \end{lstlisting}
    \end{block}

    \begin{block}{\textbf{Filter 6: Compound Multi-Filter (FilterList MUST\_PASS\_ALL)}}
        \textit{Goal: High priority query: Narcotics crimes with confirmed arrests in District 011.}
        \begin{lstlisting}[language=bash]
scan 'crime_records', {FILTER => "(PrefixFilter('011#')) AND (SingleColumnValueFilter('details', 'arrest', =, 'binary:true'))", LIMIT => 5}
        \end{lstlisting}
    \end{block}
\end{frame}

% Slide 12: Step 5 - Java API Implementation
\begin{frame}[fragile]{Step 5: HBase Java API Implementation (Rubric: 2 Marks)}
    \begin{itemize}
        \item \textbf{Class:} \texttt{bigdata.HBaseCrimeOperations.java}
        \item \textbf{Capabilities Implemented:} Table Admin, Put, Get, Filter Scan, and Delete.
    \end{itemize}
    \begin{lstlisting}[language=Java]
Configuration config = HBaseConfiguration.create();
try (Connection conn = ConnectionFactory.createConnection(config);
     Table table = conn.getTable(TableName.valueOf("crime_records"))) {
    
    // Programmatic Filtered Scan in Java API
    Scan scan = new Scan();
    SingleColumnValueFilter filter = new SingleColumnValueFilter(
        Bytes.toBytes("details"), Bytes.toBytes("arrest"),
        CompareOperator.EQUAL, new BinaryComparator(Bytes.toBytes("true"))
    );
    scan.setFilter(filter);
    scan.setLimit(5);

    try (ResultScanner scanner = table.getScanner(scan)) {
        for (Result r : scanner) {
            System.out.println("Match: " + Bytes.toString(r.getRow()));
        }
    }
}
    \end{lstlisting}
\end{frame}
