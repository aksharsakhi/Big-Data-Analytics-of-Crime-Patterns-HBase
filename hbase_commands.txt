# ------------------------------------------------------------------------------
# 6. APPLICATION-SPECIFIC ANALYTICAL QUERIES & FILTERS (RUBRIC REQUIREMENT: 4 MARKS)
# ------------------------------------------------------------------------------

# Filter 1: SingleColumnValueFilter - Find All Crimes Resulting in Arrest (Law Enforcement Efficiency)
scan 'crime_records', {FILTER => "SingleColumnValueFilter('details', 'arrest', =, 'binary:true')", LIMIT => 5}

# Filter 2: SingleColumnValueFilter - Track Domestic Violence Incidents (Family Safety Intervention)
scan 'crime_records', {FILTER => "SingleColumnValueFilter('details', 'domestic', =, 'binary:true')", LIMIT => 5}

# Filter 3: PrefixFilter - Rapid Jurisdiction Retrieval for District 019 (Patrol Deployment)
scan 'crime_records', {FILTER => "PrefixFilter('019#')", LIMIT => 5}

# Filter 4: RowFilter with RegexStringComparator - Cross-District Narcotics Incident Tracking
scan 'crime_records', {FILTER => "RowFilter(=, 'regexstring:^.*#NARCOTICS#.*')", LIMIT => 5}

# Filter 5: ValueFilter with SubstringComparator - Keyword Investigation for Vehicle Offenses
scan 'crime_records', {FILTER => "ValueFilter(=, 'substring:VEHICLE')", LIMIT => 5}

# Filter 6: Compound Multi-Filter (FilterList MUST_PASS_ALL) - High Priority: District 011 Narcotics Arrests
scan 'crime_records', {FILTER => "(PrefixFilter('011#')) AND (SingleColumnValueFilter('details', 'arrest', =, 'binary:true'))", LIMIT => 5}

# Filter 7: ColumnPaginationFilter - Paginate Attributes for Lightweight Network Transfer
scan 'crime_records', {FILTER => "ColumnPaginationFilter(4, 0)", LIMIT => 5}
